npm install -D @playwright/test

// package.json — scripts por tipo de test
{
  "scripts": {
    // ── E2E (Playwright) ─────────────────────────────────────
    "test:e2e":            "playwright test",
    "test:e2e:legacy":    "playwright test --project=legacy-code",
    "test:e2e:cart":      "playwright test --project=cartografia",
    "test:e2e:validacion":"playwright test --project=validacion",
    "test:smoke":         "playwright test --project=validacion --grep @smoke",
    "test:owasp":         "playwright test --project=validacion --grep @owasp",

    // ── Integración Back (JUnit + H2) ────────────────────────
    "test:int:back":      "mvn verify -Dtest.skip=false",
    "test:int:transport": "mvn verify -Dit.test='*ServletIT'",
    "test:int:infra":     "mvn verify -Dit.test='*DaoIT'",

    // ── Integración Front (Vitest + jsdom) ───────────────────
    "test:int:front":     "vitest run tests/integracion/front",

    // ── CI: solo lo que corre siempre ────────────────────────
    "test:ci":            "npm run test:e2e:validacion && npm run test:int:back && npm run test:int:front"
  }
}
💡 La regla de qué ejecutar y cuándo
Una sola vez: test:e2e:legacy (confirmar deploy) → test:e2e:cart (capturar HARs).
Durante el refactor: test:int:back + test:int:front en cada commit para tener feedback rápido sin navegador.
Siempre (CI): test:ci — validación e2e + integración back + integración front. Si algo aquí falla, el PR no se mergea.

PANEL TESTING — ANATOMÍA COMPLETA
▶ Run All Tests
Ejecuta toda la suite. Equivale a npx playwright test. Resultados con iconos ✓/✗/⊙ por test.
▷ Run Test (hover)
Al hacer hover sobre un test aparece ▷. Ejecuta solo ese test. Imprescindible mientras desarrollas.
👁️ Show Browser
Muestra el navegador durante la ejecución. Equivale a --headed. Actívalo para entender fallos visuales.
🎬 Record New
Abre Codegen. Graba acciones → genera código TypeScript en tiempo real. Elige el fichero destino antes de grabar.
📍 Pick Locator
Abre navegador en modo "pick". Clic en un elemento → sugiere el locator más robusto en el panel lateral.
🔁 Record at Cursor
Como Record New pero inserta las acciones justo donde está el cursor en el editor. Útil para ampliar un test existente.
🐞 Debug Test
Adjunta el debugger de VSCode. Breakpoints en TypeScript, pausa el navegador en ese punto. Ver variables en tiempo real.
📁 Reveal in Explorer
Navega al fichero del test seleccionado en el explorador. Útil cuando tienes muchos spec files.

PRIORIDAD DE SELECTORES — DE MEJOR A PEOR
Regla de oro: elige el selector que menos cambie cuando cambie el código. Roles ARIA y labels cambian solo si cambia la semántica de la UI (raramente). Clases CSS cambian con cada refactor de estilos (frecuentemente).
getByRole
page.getByRole('button', { name: 'Guardar' })
Rol ARIA + nombre accesible. El más robusto. No se rompe con cambios de CSS ni de posición en el DOM.
MEJOR
getByLabel
page.getByLabel('Dirección de email')
Input asociado a un <label>. Ideal para formularios bien construidos.
MEJOR
getByText
page.getByText('Guardar cambios', { exact: true })
Texto visible. Cuidado con textos dinámicos o i18n. Usa exact:true para evitar coincidencias parciales.
BUENO
getByPlaceholder
page.getByPlaceholder('usuario@email.com')
Atributo placeholder. Útil cuando no hay label. Menos estable si cambia el copy.
BUENO
getByAltText
page.getByAltText('Logo empresa')
Atributo alt de imágenes. El único selector semántico para <img>.
BUENO
getByTitle
page.getByTitle('Cerrar modal')
Atributo title. Menos usado pero estable para iconos y tooltips.
BUENO
getByTestId
page.getByTestId('btn-guardar-cuenta')
Atributo data-testid. Requiere añadirlo al HTML. El más explícito y estable a largo plazo.
BUENO
locator ID
page.locator('#email')
ID único. Estable si el ID no cambia, pero en legacy suelen ser técnicos o generados.
EVITAR
locator CSS clase
page.locator('.btn-primary')
Frágil. Las clases CSS cambian con cada refactor de estilos. Solo si no hay alternativa.
EVITAR
locator XPath
page.locator('//div//input[1]')
Muy frágil. Se rompe con cualquier cambio estructural del HTML. Último recurso absoluto.

// Primero, último, enésimo (0-indexed)
page.locator('tr').first()

INTERCEPTAR Y MODIFICAR REQUESTS — page.route()
page.route() es una de las funciones más potentes de Playwright. Permite interceptar cualquier request HTTP del navegador y: modificarla, devolverla desde un mock, abortarla, o dejarla pasar. Imprescindible para simular errores del servidor y para tests aislados que no dependen de la BD. 
// Mockear una respuesta completa
await page.route('**/api/cuentas', route => route.fulfill({
  status: 200,
  contentType: 'application/json',
  body: JSON.stringify([
    { email: 'mock@test.com', nombre: 'Usuario Mock' }
  ]),
}));

// Simular error del servidor (útil para testar manejo de errores)
await page.route('**/api/cuentas', route => route.fulfill({
  status: 500,
  body: 'Internal Server Error',
}));

// Simular timeout de red
await page.route('**/api/contratos', route => route.abort('timedout'));

// Dejar pasar pero modificar la respuesta
await page.route('**/api/cuentas', async route => {
  const response = await route.fetch(); // hacer el request real
  const body = await response.json();
  body.push({ email: 'extra@test.com' }); // añadir datos
  await route.fulfill({ response, body: JSON.stringify(body) });
});

// Desactivar el mock después de usarlo
await page.unroute('**/api/cuentas');
await page.unrouteAll();             // desactivar todos

import { Page, Locator, expect } from '@playwright/test';

export class CuentasPage {
  private readonly page: Page;
  readonly emailInput: Locator;
  readonly nombreInput: Locator;
  readonly passwordInput: Locator;
  readonly guardarBtn: Locator;
  readonly tablaCuentas: Locator;

  constructor(page: Page) {
    this.page          = page;
    this.emailInput    = page.locator('input[name="email"]');
    this.nombreInput   = page.locator('input[name="nombre"]');
    this.passwordInput = page.locator('input[name="password"]');
    this.guardarBtn    = page.getByRole('button', { name: 'Guardar' });
    this.tablaCuentas  = page.locator('#tabla-cuentas tbody');
  }

  async goto() { await this.page.goto('/cuentas.html'); }

  async crearCuenta(email: string, nombre: string, password: string) {
    await this.emailInput.fill(email);
    await this.nombreInput.fill(nombre);
    await this.passwordInput.fill(password);
    await this.guardarBtn.click();
  }

  async buscarCuenta(email: string): Promise<Locator> {
    return this.tablaCuentas.locator('tr')
      .filter({ hasText: email });
  }

  async expectCuentaVisible(email: string) {
    const fila = await this.buscarCuenta(email);
    await expect(fila).toBeVisible();
  }
}

FALLOS FRECUENTES Y SOLUCIONES
ErrorCausa más probableSolución
Timeout waiting for locator El selector no encuentra el elemento en el timeout Usar Pick Locator para verificar el selector. Activar Show Browser para ver qué hay en pantalla.
strict mode violation (2+ elements) El locator devuelve más de un elemento Añadir .first(), .nth(0), o refinar con .filter().
Element is not visible Existe en DOM pero hidden o fuera del viewport locator.scrollIntoViewIfNeeded(). Verificar display/visibility en DevTools.
Navigation failed / ERR_CONNECTION El servidor no está corriendo o la URL es incorrecta Verificar que el servidor arrancó. Comprobar baseURL en el config.
storageState expired La sesión guardada expiró (token caducado) Borrar auth/admin.json y ejecutar npx playwright test tests/auth.setup.ts.
Falla en CI, pasa en local Timings, headless vs headed, resolución diferente Añadir retries: 2 en CI. Usar waitForSelector explícito. Revisar la traza con Trace Viewer.
intercept no funciona La URL del route no coincide con el patrón Loguear las URLs reales: page.on('request', r => console.log(r.url())).
Test flaky (falla 1 de 10 veces) Race condition: el test actúa antes de que la UI esté lista Añadir waitForResponse o waitForSelector antes de la acción que falla.

// scripts/extract-contracts.mjs
// Uso: node extract-contracts.mjs telemetry/har/iter-02/busqueda.har
import fs from 'fs';
import path from 'path';

const harFile = process.argv[2];
if (!harFile) { console.error('Uso: node extract-contracts.mjs fichero.har'); process.exit(1); }

const har = JSON.parse(fs.readFileSync(harFile, 'utf-8'));

const contracts = har.log.entries
  .filter(e => {
    const url = e.request.url;
    return url.includes('/api/') || url.includes('.jsp');
  })
  .map(e => ({
    endpoint:        new URL(e.request.url).pathname,
    method:          e.request.method,
    queryString:     e.request.queryString || [],
    requestBody:     e.request.postData?.text || null,
    responseStatus:  e.response.status,
    responseHeaders: e.response.headers
      .filter(h => ['content-type','location','set-cookie','x-frame-options']
        .includes(h.name.toLowerCase()))
      .reduce((acc, h) => ({ ...acc, [h.name]: h.value }), {}),
    responseBody:    e.response.content?.text || null,
    timingMs:        Math.round(e.time),
  }));

// Deduplicar por endpoint + método + responseStatus
const seen = new Set<string>();
const unique = contracts.filter(c => {
  const key = `${c.method}:${c.endpoint}:${c.responseStatus}`;
  if (seen.has(key)) return false;
  seen.add(key); return true;
});

const output = harFile.replace('.har', '-contracts.json');
fs.writeFileSync(output, JSON.stringify(unique, null, 2));
console.log(`✓ ${unique.length} contratos únicos → ${output}`);

// tests/e2e-seguridad/owasp.spec.ts
import { test, expect } from '@playwright/test';

test.describe('OWASP Top 10 — post-migración @smoke', () => {

  test('A01 — endpoint sin sesión → 401/403/302', async ({ request }) => {
    const res = await request.get('/api/cuentas', { headers: { Cookie: '' } });
    expect([401, 403, 302]).toContain(res.status());
  });

  test('A03 — SQL injection rechazado sin filtrar datos', async ({ request }) => {
    const res = await request.get("/api/cuentas?email=' OR '1'='1");
    const body = await res.text();
    expect(body).not.toContain('admin@');
    expect(res.status()).not.toBe(500); // no revienta la BD
  });

  test('A05 — todos los security headers presentes', async ({ request }) => {
    const res = await request.get('/');
    const h = res.headers();
    expect.soft(h['x-content-type-options']).toBe('nosniff');
    expect.soft(h['x-frame-options']).toBe('DENY');
    expect.soft(h['strict-transport-security']).toBeDefined();
    expect.soft(h['content-security-policy']).toBeDefined();
    expect.soft(h['referrer-policy']).toBeDefined();
  });

  test('A05 — errores no exponen stack traces', async ({ request }) => {
    const res = await request.get('/api/no-existe');
    const body = await res.text();
    expect(body).not.toContain('at java.');
    expect(body).not.toContain('Exception');
    expect(body).not.toContain('NullPointer');
  });

  test('A07 — cookie tiene Secure + HttpOnly', async ({ request }) => {
    const res = await request.post('/login', {
      data: { usuario: 'admin', password: 'test1234' }
    });
    const cookie = (res.headers()['set-cookie'] || '').toLowerCase();
    expect.soft(cookie).toContain('httponly');
    expect.soft(cookie).toContain('secure');
    expect.soft(cookie).toContain('samesite');
  });

  test('A02 — password no aparece en ninguna response', async ({ page }) => {
    const responses: string[] = [];
    page.on('response', async res => {
      if (res.url().includes('/api/')) {
        const body = await res.text().catch(() => '');
        responses.push(body);
      }
    });
    await page.goto('/');
    await page.waitForLoadState('networkidle');
    const joined = responses.join(' ');
    expect(joined).not.toContain('"password":');
    expect(joined).not.toContain('"passwordHash":');
  });
});
