⚠ Transacciones por chunk: Si falla en write() → rollback de los N ítems completos. El Reader es stateful → sabe dónde reanudar tras restart.
Comparativa: Chunk vs Tasklet
Chunk (procesarPedidosStep): Procesa muchos ítems en lotes con Reader/Processor/Writer. Transacción por cada chunk. Ideal para procesar millones de registros.

Tasklet (validarArchivoStep, notificarStep): Ejecuta una tarea única con execute(). Una sola transacción. Ideal para validaciones, limpieza, notificaciones.
Código completo
@Configuration
public class BatchConfig {

  @Bean
  public Job importarPedidosJob(JobRepository repo,
                                    Step validarArchivoStep,
                                    Step procesarPedidosStep,
                                    Step notificarStep) {
    return new JobBuilder("importarPedidosJob", repo)
      .start(validarArchivoStep)      // Step 1: Tasklet
      .next(procesarPedidosStep)      // Step 2: Chunk
      .next(notificarStep)            // Step 3: Tasklet
      .build();
  }

  @Bean
  public Step validarArchivoStep(JobRepository repo,
                                  PlatformTransactionManager tx) {
    return new StepBuilder("validarArchivoStep", repo)
      .tasklet(new FileCheckTasklet(), tx)
      .build();
  }

  @Bean
  public Step procesarPedidosStep(JobRepository repo,
                                    PlatformTransactionManager tx,
                                    ItemReader<Pedido> reader,
                                    ItemProcessor<Pedido, PedidoDTO> processor,
                                    ItemWriter<PedidoDTO> writer) {
    return new StepBuilder("procesarPedidosStep", repo)
      .<Pedido, PedidoDTO>chunk(100, tx)
      .reader(reader)
      .processor(processor)
      .writer(writer)
      .build();
  }

  @Bean
  public Step notificarStep(JobRepository repo,
                              PlatformTransactionManager tx) {
    return new StepBuilder("notificarStep", repo)
      .tasklet(new EmailNotifierTasklet(), tx)
      .build();
  }
}
public class FileCheckTasklet implements Tasklet {

  @Override
  public RepeatStatus execute(
      StepContribution contribution,
      ChunkContext chunkContext) throws Exception {

    File file = new File("pedidos.csv");

    if (!file.exists()) {
      throw new FileNotFoundException("CSV no encontrado");
    }

    System.out.println("Archivo validado OK");
    return RepeatStatus.FINISHED;
  }
}

// Diferencia clave con Chunk:
// - execute() se llama UNA sola vez
// - No hay Reader/Processor/Writer
// - Toda la lógica va aquí dentro
// - Una sola transacción para todo
@Bean
public FlatFileItemReader<Pedido> pedidoReader() {
  return new FlatFileItemReaderBuilder<Pedido>()
    .name("pedidoReader")
    .resource(new FileSystemResource("pedidos.csv"))
    .delimited().delimiter(",")
    .names("id", "cliente", "importe")
    .targetType(Pedido.class)
    .build();
}

// read() devuelve:
//   → un Pedido  si quedan líneas
//   → null       cuando se acaba el fichero (señal de FIN)
@Component
public class PedidoItemProcessor
    implements ItemProcessor<Pedido, PedidoDTO> {

  @Override
  public PedidoDTO process(Pedido pedido) {

    // Filtrar: devolver null descarta el ítem
    if (pedido.getImporte() <= 0) return null;

    // Transformar
    PedidoDTO dto = new PedidoDTO();
    dto.setId(pedido.getId());
    dto.setImporteConIva(pedido.getImporte() * 1.21);
    return dto;
  }
}
@Bean
public JdbcBatchItemWriter<PedidoDTO> pedidoWriter(DataSource ds) {
  return new JdbcBatchItemWriterBuilder<PedidoDTO>()
    .dataSource(ds)
    .sql("INSERT INTO pedidos (id, importe_iva) VALUES (:id, :importeConIva)")
    .beanMapped()
    .build();
}

// El Writer recibe List<PedidoDTO> (tamaño = chunk)
