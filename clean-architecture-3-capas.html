⛔ La regla que no puedes romper

Si tu entidad tiene @Entity de JPA, o tu Use Case hace import javax.mail, la inversión se rompe. El dominio depende de persistencia y ya no puedes migrar sin reescribir reglas de negocio.

// domain/entities/FormSubmission.java — Bean puro
public class FormSubmission {
    private final String id, senderName, senderEmail, subject, body;
    private String status = "PENDING";
    private final LocalDateTime createdAt = LocalDateTime.now();
    private LocalDateTime processedAt;

    public FormSubmission(String id, String name, String email,
                          String subject, String body) {
        this.id=id; this.senderName=name; this.senderEmail=email;
        this.subject=subject; this.body=body;
    }

    public void markAsProcessed() {
        status = "PROCESSED"; processedAt = LocalDateTime.now();
    }
    public void markAsSent() {
        if (!"PROCESSED".equals(status))
            throw new IllegalStateException("No procesado");
        status = "SENT";
    }
    public void markAsFailed() { status = "FAILED"; }

    // Getters (sin setters — inmutabilidad controlada)
    public String getId(){return id;} public String getSenderEmail(){return senderEmail;}
    public String getSubject(){return subject;} public String getBody(){return body;}
    public String getStatus(){return status;} public String getSenderName(){return senderName;}
}

Use Case

ProcessAndSendFormUseCase — Recibe Repositories por constructor

Orquesta: procesar → guardar → generar doc → enviar → actualizar. No conoce ninguna tecnología.

▶
// domain/usecases/ProcessAndSendFormUseCase.java
public class ProcessAndSendFormUseCase {
    private final FormRepository formRepo;
    private final DocumentRepository docRepo;
    private final MailRepository mailRepo;

    public ProcessAndSendFormUseCase(FormRepository f,
            DocumentRepository d, MailRepository m) {
        formRepo=f; docRepo=d; mailRepo=m;
    }

    public void execute(FormSubmission form, String templateId) {
        form.markAsProcessed();
        formRepo.save(form);
        try {
            byte[] pdf = docRepo.generate(form, templateId);
            mailRepo.send(form.getSenderEmail(),
                "Confirmación: "+form.getSubject(),
                "Adjuntamos su formulario procesado.", pdf);
            form.markAsSent();
        } catch(Exception e) { form.markAsFailed(); }
        formRepo.save(form);
    }
}
// domain/usecases/ProcessAndSendFormUseCase.ts
export class ProcessAndSendFormUseCase {
  constructor(private formRepo:FormRepository,
    private docRepo:DocumentRepository,
    private mailRepo:MailRepository) {}

  async execute(form:FormSubmission, templateId:string) {
    form.markAsProcessed();
    await this.formRepo.save(form);
    try {
      const pdf = await this.docRepo.generate(form, templateId);
      await this.mailRepo.send(form.senderEmail,
        `Confirmación: ${form.subject}`,
        "Adjuntamos su formulario.", pdf);
      form.markAsSent();
    } catch { form.markAsFailed(); }
    await this.formRepo.save(form);
  }
}
// domain/process_and_send.go
func (uc *ProcessAndSendFormUseCase) Execute(
    f *FormSubmission, tid string) error {
    f.MarkAsProcessed()
    if err:=uc.formRepo.Save(f);err!=nil{return err}
    pdf,err:=uc.docRepo.Generate(f,tid)
    if err!=nil{f.MarkAsFailed();uc.formRepo.Save(f);return err}
    if err=uc.mailRepo.Send(f.SenderEmail,
        "Confirmación: "+f.Subject,
        "Adjuntamos su formulario.",pdf);err!=nil{
        f.MarkAsFailed()
    }else{f.MarkAsSent()}
    return uc.formRepo.Save(f)
}
💡 Lógica idéntica en 3 lenguajes

markAsProcessed → save → generate → send → markAsSent/Failed → save. Solo cambia la sintaxis. Portable porque solo depende de interfaces Repository.

// persistence/JdbcFormRepository.java — implements FormRepository
public class JdbcFormRepository implements FormRepository {
    private final Connection conn;
    public JdbcFormRepository(Connection c) { conn=c; }

    @Override public void save(FormSubmission f) {
        String sql = "INSERT INTO forms(id,sender_name,sender_email,"
            + "subject,body,status) VALUES(?,?,?,?,?,?) "
            + "ON DUPLICATE KEY UPDATE status=?";
        try(PreparedStatement ps = conn.prepareStatement(sql)) {
            ps.setString(1, f.getId()); ps.setString(2, f.getSenderName());
            ps.setString(3, f.getSenderEmail()); ps.setString(4, f.getSubject());
            ps.setString(5, f.getBody()); ps.setString(6, f.getStatus());
            ps.setString(7, f.getStatus()); ps.executeUpdate();
        } catch(SQLException e) { throw new RuntimeException(e); }
    }
}
// persistence/PgFormRepository.ts — implements FormRepository
export class PgFormRepository implements FormRepository {
  constructor(private pool: Pool) {}
  async save(f: FormSubmission) {
    await this.pool.query(
      `INSERT INTO forms(id,sender_name,sender_email,subject,body,status)
       VALUES($1,$2,$3,$4,$5,$6)
       ON CONFLICT(id) DO UPDATE SET status=$6`,
      [f.id, f.senderName, f.senderEmail, f.subject, f.body, f.status]);
  }
}
// persistence/pg_form_repo.go — implements FormRepository
func (r *PgFormRepository) Save(f *domain.FormSubmission) error {
    _, err := r.db.Exec(`INSERT INTO forms(id,sender_name,
        sender_email,subject,body,status) VALUES($1,$2,$3,$4,$5,$6)
        ON CONFLICT(id) DO UPDATE SET status=$6`,
        f.ID, f.SenderName, f.SenderEmail, f.Subject, f.Body, f.Status)
    return err
}
🔌 El punto clave del patrón Repository

Las 3 implementaciones satisfacen la misma interfaz FormRepository. El Use Case no cambia. El controller no cambia. Solo cambia qué Repository se inyecta en el main().

// persistence/FopDocumentRepository.java
public class FopDocumentRepository implements DocumentRepository {
    private final String templatesDir;
    public FopDocumentRepository(String dir) { templatesDir=dir; }

    @Override
    public byte[] generate(FormSubmission data, String templateId) {
        String xml = toXml(data);
        File xsl = new File(templatesDir, templateId+".xsl");
        FopFactory ff = FopFactory.newInstance(new File(".").toURI());
        ByteArrayOutputStream out = new ByteArrayOutputStream();
        Fop fop = ff.newFop(MimeConstants.MIME_PDF, out);
        Transformer t = TransformerFactory.newInstance()
            .newTransformer(new StreamSource(xsl));
        t.transform(new StreamSource(new StringReader(xml)),
                    new SAXResult(fop.getDefaultHandler()));
        return out.toByteArray();
    }
}
// persistence/XsltDocumentRepository.ts
export class XsltDocumentRepository implements DocumentRepository {
  constructor(private templatesDir: string) {}
  async generate(data:FormSubmission, tid:string): Promise<Buffer> {
    const xml = this.toXml(data);
    const tmp = `/tmp/${data.id}`;
    writeFileSync(`${tmp}.xml`, xml);
    execSync(`xsltproc -o ${tmp}.fo ${this.templatesDir}/${tid}.xsl ${tmp}.xml`);
    execSync(`fop ${tmp}.fo ${tmp}.pdf`);
    const pdf = readFileSync(`${tmp}.pdf`);
    [".xml",".fo",".pdf"].forEach(e=>unlinkSync(tmp+e));
    return pdf;
  }
}

// transport/FormServlet.java
public class FormServlet extends HttpServlet {
    private final ProcessAndSendFormUseCase useCase;
    public FormServlet(ProcessAndSendFormUseCase uc){useCase=uc;}

    @Override protected void doPost(HttpServletRequest req,
                                       HttpServletResponse resp) {
        var form = new FormSubmission(UUID.randomUUID().toString(),
            req.getParameter("name"), req.getParameter("email"),
            req.getParameter("subject"), req.getParameter("body"));
        String t = req.getParameter("template");
        useCase.execute(form, t!=null?t:"informe-formal");
        resp.setStatus(201);
    }
}
// transport/formRouter.ts — todo lo de Express
export function createFormRouter(uc:ProcessAndSendFormUseCase) {
  const r = Router();
  r.post("/api/forms", async(req,res)=>{
    const {name,email,subject,body,template} = req.body;
    const form = new FormSubmission(uuid(),name,email,subject,body);
    await uc.execute(form, template??"informe-formal");
    res.status(201).json({id:form.id, status:form.status});
  });
  return r;
}
// transport/form_handler.go
func(h *FormHandler)HandleSubmit(c *gin.Context){

    c.BindJSON(&req)
    form:=domain.NewFormSubmission(uuid.New().String(),
        req.Name,req.Email,req.Subject,req.Body)
    t:=req.Template; if t==""{t="informe-formal"}
    h.uc.Execute(form,t)
    c.JSON(201,gin.H{"id":form.ID})
}
