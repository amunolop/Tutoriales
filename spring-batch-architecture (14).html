Chunk vs Tasklet — ¿Cuándo usar cada uno?

Chunk (procesarPedidosStep): Procesa muchos ítems en lotes con Reader/Processor/Writer. Transacción por cada chunk. Ideal para procesar millones de registros. Loop automático hasta que Reader devuelva null.

Tasklet (validarArchivoStep, notificarStep): Ejecuta una tarea única con execute(). Una sola transacción. Ideal para validaciones, limpieza, notificaciones, llamadas a APIs. Se ejecuta una sola vez.
JobRepository — el cerebro persistente
🗄️
JobRepository
Persiste el estado de cada ejecución en BD. Permite reiniciar desde el punto de fallo sin reprocesar lo ya completado. Cada vez que lanzas un Job crea una nueva JobInstance si los parámetros son distintos, o reutiliza la existente para continuar.

1. Spring escanea @Configuration └─> Encuentra BatchConfig.java 2. Spring ejecuta métodos @Bean (orden automático por dependencias): a) pedidoReader() → crea FlatFileItemReader └─> Bean guardado en contexto b) pedidoProcessor() → crea PedidoItemProcessor └─> Bean guardado en contexto c) pedidoWriter(DataSource ds) └─> Spring inyecta DataSource automáticamente └─> crea JdbcBatchItemWriter └─> Bean guardado en contexto d) validarArchivoStep(JobRepository repo, TransactionManager tx) └─> Spring inyecta repo + tx └─> crea StepBuilder con new FileCheckTasklet() └─> Bean Step guardado e) procesarPedidosStep(repo, tx, reader, processor, writer) └─> Spring inyecta los 5 parámetros (todos ya creados en a, b, c) └─> crea StepBuilder con .chunk(100, tx) └─> enlaza reader → processor → writer └─> Bean Step guardado f) notificarStep(repo, tx) └─> crea StepBuilder con new EmailNotifierTasklet() └─> Bean Step guardado g) importarPedidosJob(repo, step1, step2, step3) └─> Spring inyecta repo + los 3 Steps └─> crea JobBuilder └─> enlaza .start(step1).next(step2).next(step3) └─> Bean Job guardado 3. Spring Boot detecta job.enabled=true └─> JobLauncherApplicationRunner arranca

Ejecuta Job.execute() │ │ │ ├─> STEP 1: validarArchivoStep (Tasklet) │ │ ├─> Crea StepExecution → INSERT en BATCH_STEP_EXECUTION │ │ ├─> Inicia transacción │ │ ├─> Llama a FileCheckTasklet.execute() │ │ │ └─> File file = new File("pedidos.csv") │ │ │ └─> if (!file.exists()) throw Exception │ │ │ └─> return RepeatStatus.FINISHED │ │ ├─> COMMIT transacción │ │ └─> UPDATE BATCH_STEP_EXECUTION (status=COMPLETED) │ │ │ ├─> STEP 2: procesarPedidosStep (Chunk) │ │ ├─> Crea StepExecution → INSERT en BATCH_STEP_EXECUTION │ │ ├─> Reader.open() → FlatFileItemReader abre pedidos.csv │ │ │ │ │ ├─> ──────── LOOP CHUNK ──────── │ │ │ ├─> Inicia transacción │ │ │ ├─> READ × 100: reader.read() → List<Pedido> │ │ │ ├─> PROCESS × 100: processor.process() → List<PedidoDTO> │ │ │ ├─> WRITE: writer.write(List) → batch INSERT en BD │ │ │ ├─> COMMIT → guarda posición en BATCH_STEP_EXECUTION_CONTEXT │ │ │ └─> ¿read() == null? → NO: siguiente chunk · SÍ: salir │ │ │ │ │ ├─> Reader.close() │ │ └─> UPDATE BATCH_STEP_EXECUTION (status=COMPLETED) │ │ │ └─> STEP 3: notificarStep (Tasklet) │ ├─> Crea StepExecution │ ├─> Inicia transacción │ ├─> Llama a EmailNotifierTasklet.execute() │ │ └─> Lee StepExecutionContext del Step 2 │ │ └─> Obtiene métricas (readCount, writeCount) │ │ └─> Envía email con resumen │ │ └─> return RepeatStatus.FINISHED │ ├─> COMMIT transacción │ └─> UPDATE BATCH_STEP_EXECUTION (status=COMPLETED) │ └─> 3. Job termina └─> UPDATE BATCH_JOB_EXECUTION (status=COMPLETED, endTime=...) └─> JobLauncher.run() retorna JobExecution
Punto clave del chunk loop: Si el Writer falla, se hace rollback de los 100 ítems completos. El Reader reanuda desde la última posición confirmada en BATCH_STEP_EXECUTION_CONTEXT, no desde el principio.

@Configuration
public class BatchConfig {

  @Bean
  public Job importarPedidosJob(JobRepository repo,
                                    Step validarArchivoStep,
                                    Step procesarPedidosStep,
                                    Step notificarStep) {
    return new JobBuilder("importarPedidosJob", repo)
      .start(validarArchivoStep)      // Step 1: Tasklet
      .next(procesarPedidosStep)      // Step 2: Chunk
      .next(notificarStep)            // Step 3: Tasklet
      .build();
  }

  @Bean
  public Step validarArchivoStep(JobRepository repo,
                                  PlatformTransactionManager tx) {
    return new StepBuilder("validarArchivoStep", repo)
      .tasklet(new FileCheckTasklet(), tx)
      .build();
  }

  @Bean
  public Step procesarPedidosStep(JobRepository repo,
                                    PlatformTransactionManager tx,
                                    ItemReader<Pedido> reader,
                                    ItemProcessor<Pedido, PedidoDTO> processor,
                                    ItemWriter<PedidoDTO> writer) {
    return new StepBuilder("procesarPedidosStep", repo)
      .<Pedido, PedidoDTO>chunk(100, tx)
      .reader(reader)
      .processor(processor)
      .writer(writer)
      .build();
  }

  @Bean
  public Step notificarStep(JobRepository repo,
                              PlatformTransactionManager tx) {
    return new StepBuilder("notificarStep", repo)
      .tasklet(new EmailNotifierTasklet(), tx)
      .build();
  }
}
public class FileCheckTasklet implements Tasklet {

  @Override
  public RepeatStatus execute(
      StepContribution contribution,
      ChunkContext chunkContext) throws Exception {

    File file = new File("pedidos.csv");

    if (!file.exists()) {
      throw new FileNotFoundException("CSV no encontrado");
    }

    System.out.println("Archivo validado OK");
    return RepeatStatus.FINISHED;
  }
}

@Bean
public FlatFileItemReader<Pedido> pedidoReader() {
  return new FlatFileItemReaderBuilder<Pedido>()
    .name("pedidoReader")
    .resource(new ClassPathResource("pedidos.csv"))
    .delimited().delimiter(",")
    .names("id", "cliente", "importe")
    .targetType(Pedido.class)
    .build();
}

// read() devuelve:
//   → un Pedido  si quedan líneas
//   → null       cuando se acaba el fichero (señal de FIN del chunk loop)
@Component
public class PedidoItemProcessor
    implements ItemProcessor<Pedido, PedidoDTO> {

  @Override
  public PedidoDTO process(Pedido pedido) {

    // Filtrar: devolver null descarta el ítem del chunk
    if (pedido.getImporte() <= 0) return null;

    PedidoDTO dto = new PedidoDTO();
    dto.setId(pedido.getId());
    dto.setCliente(pedido.getCliente());
    dto.setImporteConIva(pedido.getImporte() * 1.21);
    return dto;
  }
}
@Bean
public JdbcBatchItemWriter<PedidoDTO> pedidoWriter(DataSource ds) {
  return new JdbcBatchItemWriterBuilder<PedidoDTO>()
    .dataSource(ds)
    .sql("INSERT INTO pedidos (id, cliente, importe_iva) " +
         "VALUES (:id, :cliente, :importeConIva)")
    .beanMapped()
    .build();
}

// El Writer recibe List<PedidoDTO> (tamaño = chunk)
// Ejecuta un batch INSERT en una sola llamada JDBC
// El COMMIT ocurre DESPUÉS de que write() retorne sin excepción

// Sin Spring Boot — todo manual
// Equivalente exacto a lo que hace SpringApplication.run() por dentro

public class ManualApplication {

  public static void main(String[] args) throws Exception {

    // 1. Cargar application.properties manualmente
    Properties props = new Properties();
    props.load(new FileInputStream(
      "src/main/resources/application.properties"));

    // 2. Crear el contenedor vacío
    AnnotationConfigApplicationContext ctx =
      new AnnotationConfigApplicationContext();

    // 3. Registrar properties en el entorno
    ctx.getEnvironment().getPropertySources()
      .addFirst(new PropertiesPropertySource("myProps", props));

    // 4. Escanear paquetes — encuentra BatchConfig, PedidoItemProcessor
    ctx.scan("com.ejemplo");

    // 5. Arrancar el contenedor — crea todos los @Bean
    ctx.refresh();

    // 6. Crear DataSource manualmente
    DataSource ds = new DriverManagerDataSource(
      props.getProperty("spring.datasource.url"),
      props.getProperty("spring.datasource.username"),
      props.getProperty("spring.datasource.password")
    );

    // 7. Crear TransactionManager
    PlatformTransactionManager tx =
      new DataSourceTransactionManager(ds);

    // 8. Crear JobRepository
    JobRepositoryFactoryBean repoFactory =
      new JobRepositoryFactoryBean();
    repoFactory.setDataSource(ds);
    repoFactory.setTransactionManager(tx);
    repoFactory.afterPropertiesSet();
    JobRepository repo = repoFactory.getObject();

    // 9. Crear JobLauncher
    TaskExecutorJobLauncher launcher =
      new TaskExecutorJobLauncher();
    launcher.setJobRepository(repo);
    launcher.afterPropertiesSet();

    // 10. Obtener Job del contenedor
    Job job = ctx.getBean(Job.class);

    // 11. Lanzar
    JobExecution execution =
      launcher.run(job, new JobParameters());
    System.out.println("Estado: " + execution.getStatus());

    // 12. Cerrar
    ctx.close();
  }
}

  JobController(JobService jobService) {
    this.jobService = jobService;
  }

  @PostMapping("/lanzar")

// ── JobService — centraliza la lógica de lanzamiento ─────
@Service
class JobService {

  private final JobLauncher jobLauncher;
  private final Job job;

  JobService(JobLauncher jobLauncher, Job job) {
    this.jobLauncher = jobLauncher;
    this.job = job;
  }

  public String lanzar() throws Exception {
    JobParameters params = new JobParametersBuilder()
      .addLong("time", System.currentTimeMillis())
      .toJobParameters();
    JobExecution execution = jobLauncher.run(job, params);
    return "Estado: " + execution.getStatus();
  }
}

// BatchScheduler — ejecuta el Job cada día a las 23:00

@Component
public class BatchScheduler {

  private final JobLauncher jobLauncher;
  private final Job job;

  public BatchScheduler(JobLauncher jobLauncher, Job job) {
    this.jobLauncher = jobLauncher;
    this.job = job;
  }

  @Scheduled(cron = "0 0 23 * * *") // seg min hora día mes díaSemana
  public void ejecutarA23h() throws Exception {
    JobParameters params = new JobParametersBuilder()
      .addLong("time", System.currentTimeMillis())
      .toJobParameters();
    JobExecution execution = jobLauncher.run(job, params);
    System.out.println("[CRON] Estado: " + execution.getStatus());
  }
}
