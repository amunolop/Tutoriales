Decisiones de diseño
1

Escritura síncrona post-respuesta

La escritura en Redis ocurre después de que el cliente tiene sus bytes. El hilo sigue vivo pero el usuario ya recibió su respuesta. Los ~1-2ms de Redis no afectan latencia percibida. Sin threads extra, sin colas, sin complejidad.

2

Caffeine en lugar de ConcurrentHashMap crudo

Un mapa estático sin evicción crece indefinidamente hasta OutOfMemoryError. Caffeine añade límite LRU y TTL local con una línea de configuración, además de métricas de hit/miss listas para Prometheus.

3

Degradación elegante si Redis falla

Si Redis no responde, la aplicación sigue funcionando con L1. El estado no se pierde en ese nodo, solo si el propio nodo reinicia. El try/catch captura el error, lo loggea, y no rompe el flujo del usuario.

4

Transparencia total para el código de negocio

El Servlet solo hace getAttribute("userSession") y setAttribute. No conoce Redis, Caffeine ni la arquitectura distribuida. Código legado migra con cambios mínimos.

Ciclo exacto en el Filter
t=0 · Request
Filter intercepta. Obtiene sessionId de la cookie HTTP.
t=0.01ms · L1 Check
l1Cache.getIfPresent(sessionId) → HIT devuelve objeto en RAM local.
t=1-2ms · L2 Fallback (solo MISS)

t=Xms · Lógica de negocio
Servlet ejecuta. Lee y escribe en request.getAttribute("userSession").
t=Xms · ✅ CLIENTE RECIBE RESPUESTA
chain.doFilter() completa. HTTP response commiteado. Usuario tiene sus datos.

t=X+1ms · Hilo liberado

package com.ejemplo.mdss.model;

import java.io.Serializable;
import java.time.Instant;

/**
 * Objeto de sesión del usuario.
 * Es lo que viaja entre L1 (Caffeine) y L2 (Redis).
 *
 * Mantenlo plano y serializable: sin referencias circulares,
 * sin objetos pesados. Todo lo que necesite el servlet para
 * procesar la petición sin tocar base de datos.
 */
public class UserSession implements Serializable {

    private String userId;
    private String username;
    private String rol;
    private String ultimaPagina;
    private Instant ultimoAcceso;

    public UserSession() {}

    public UserSession(String userId, String username, String rol) {
        this.userId       = userId;
        this.username     = username;
        this.rol          = rol;
        this.ultimoAcceso = Instant.now();
    }

    public String getUserId()              { return userId; }
    public void   setUserId(String v)      { this.userId = v; }
    public String getUsername()            { return username; }
    public void   setUsername(String v)    { this.username = v; }
    public String getRol()                 { return rol; }
    public void   setRol(String v)         { this.rol = v; }
    public String getUltimaPagina()        { return ultimaPagina; }
    public void   setUltimaPagina(String v){ this.ultimaPagina = v; }
    public Instant getUltimoAcceso()       { return ultimoAcceso; }
    public void    setUltimoAcceso(Instant v){ this.ultimoAcceso = v; }

    public void actualizarAcceso() { this.ultimoAcceso = Instant.now(); }

    @Override
    public String toString() {
        return "UserSession{userId='" + userId + "', username='" + username
                + "', rol='" + rol + "', ultimoAcceso=" + ultimoAcceso + "}";
    }
}
package com.ejemplo.mdss.cache;

import com.ejemplo.mdss.model.UserSession;
import com.fasterxml.jackson.databind.ObjectMapper;
import com.fasterxml.jackson.datatype.jsr310.JavaTimeModule;
import com.github.benmanes.caffeine.cache.Cache;
import com.github.benmanes.caffeine.cache.Caffeine;
import redis.clients.jedis.Jedis;
import redis.clients.jedis.JedisPool;
import redis.clients.jedis.JedisPoolConfig;

import java.util.concurrent.TimeUnit;
import java.util.logging.Level;
import java.util.logging.Logger;

/**
 * NÚCLEO DE LA ARQUITECTURA MDSS
 * ================================
 * Gestiona el estado de sesión en dos capas:
 *
 *   L1 → Caffeine (RAM local, ~0ms, max 5000 sesiones, TTL 30 min)
 *   L2 → Redis    (RAM compartida, ~1-2ms, TTL 30 min)
 *
 * Flujo de LECTURA:
 *   L1 HIT  → devuelve directo desde RAM local
 *   L1 MISS → busca en Redis → rehidrata L1 → devuelve
 *
 * Flujo de ESCRITURA (post-respuesta HTTP):
 *   1. Actualiza L1
 *   2. Persiste en Redis de forma síncrona
 *      (el cliente ya recibió la respuesta, no afecta latencia percibida)
 */
public class SessionStateManager {

    private static final Logger LOG = Logger.getLogger(SessionStateManager.class.getName());
    private static final int    TTL_SEGUNDOS = 30 * 60;
    private static final String REDIS_PREFIX = "session:";

    private final Cache<String, UserSession> l1Cache;
    private final JedisPool jedisPool;
    private final ObjectMapper objectMapper;

    public SessionStateManager(String redisHost, int redisPort) {
        this.l1Cache = Caffeine.newBuilder()
                .maximumSize(5_000)
                .expireAfterWrite(TTL_SEGUNDOS, TimeUnit.SECONDS)
                .recordStats()
                .build();

        JedisPoolConfig cfg = new JedisPoolConfig();
        cfg.setMaxTotal(20);
        cfg.setMaxIdle(5);
        cfg.setTestOnBorrow(true);
        this.jedisPool = new JedisPool(cfg, redisHost, redisPort, 2000);

        this.objectMapper = new ObjectMapper();
        this.objectMapper.registerModule(new JavaTimeModule());

        LOG.info("SessionStateManager iniciado → Redis: " + redisHost + ":" + redisPort);
    }

    // ── LECTURA: L1 → L2 → null ─────────────────────────────────────────────

    public UserSession obtenerSesion(String sessionId) {
        UserSession sesion = l1Cache.getIfPresent(sessionId);
        if (sesion != null) {
            LOG.fine("L1 HIT → " + sessionId);
            return sesion;
        }

        LOG.fine("L1 MISS → Redis: " + sessionId);
        sesion = obtenerDesdeRedis(sessionId);
        if (sesion != null) {
            l1Cache.put(sessionId, sesion);
        }
        return sesion;
    }

    // ── ESCRITURA: L1 + L2 (llamar DESPUÉS de enviar la respuesta HTTP) ──────

    /**
     * IMPORTANTE: llamar DESPUÉS de chain.doFilter().
     * El cliente ya recibió su respuesta HTTP.
     * La escritura en Redis (~1-2ms) no afecta latencia percibida.
     */
    public void persistirSesion(String sessionId, UserSession sesion) {
        sesion.actualizarAcceso();
        l1Cache.put(sessionId, sesion);
        persistirEnRedis(sessionId, sesion);
    }

    public void eliminarSesion(String sessionId) {
        l1Cache.invalidate(sessionId);
        eliminarDeRedis(sessionId);
    }

    // ── OPERACIONES REDIS ────────────────────────────────────────────────────

    private UserSession obtenerDesdeRedis(String sessionId) {
        try (Jedis jedis = jedisPool.getResource()) {
            String json = jedis.get(REDIS_PREFIX + sessionId);
            if (json == null) return null;
            return objectMapper.readValue(json, UserSession.class);
        } catch (Exception e) {
            // Degradación elegante: sin Redis seguimos con L1
            LOG.log(Level.WARNING, "Error leyendo Redis: " + sessionId, e);
            return null;
        }
    }

    private void persistirEnRedis(String sessionId, UserSession sesion) {
        try (Jedis jedis = jedisPool.getResource()) {
            String json = objectMapper.writeValueAsString(sesion);
            jedis.setex(REDIS_PREFIX + sessionId, TTL_SEGUNDOS, json);
        } catch (Exception e) {
            // Sesión en L1 pero en riesgo si el nodo muere
            LOG.log(Level.SEVERE, "Error escribiendo Redis: " + sessionId, e);
        }
    }

    private void eliminarDeRedis(String sessionId) {
        try (Jedis jedis = jedisPool.getResource()) {
            jedis.del(REDIS_PREFIX + sessionId);
        } catch (Exception e) {
            LOG.log(Level.WARNING, "Error eliminando Redis: " + sessionId, e);
        }
    }

    public String obtenerEstadisticasL1() { return l1Cache.stats().toString(); }
    public void cerrar() { jedisPool.close(); }
}
package com.ejemplo.mdss.filter;

import com.ejemplo.mdss.cache.SessionStateManager;
import com.ejemplo.mdss.model.UserSession;
import jakarta.servlet.*;
import jakarta.servlet.annotation.WebFilter;
import jakarta.servlet.http.HttpServletRequest;

import java.io.IOException;
import java.util.logging.Logger;

/**
 * FILTER MDSS — El orquestador de la arquitectura
 * ================================================
 * Intercepta TODAS las peticiones HTTP y ejecuta el ciclo:
 *
 *   PRE   → carga sesión desde L1 o L2 → inyecta en request
 *   CHAIN → el Servlet ejecuta lógica de negocio
 *   POST  → response ya commiteado → escritura síncrona en Redis
 *
 * El Servlet solo necesita:
 *   leer:    request.getAttribute("userSession")
 *   escribir: request.setAttribute("userSession", sesionModificada)
 *
 * Transparencia total: el código de negocio no sabe nada de Redis.
 */
@WebFilter("/*")
public class MdssSessionFilter implements Filter {

    private static final Logger LOG =
            Logger.getLogger(MdssSessionFilter.class.getName());

    private SessionStateManager stateManager;

    @Override
    public void init(FilterConfig cfg) {
        this.stateManager = (SessionStateManager)
                cfg.getServletContext().getAttribute("stateManager");
        LOG.info("MdssSessionFilter inicializado.");
    }

    @Override
    public void doFilter(ServletRequest request, ServletResponse response,
                         FilterChain chain) throws IOException, ServletException {

        HttpServletRequest req = (HttpServletRequest) request;

        // PASO 1: Identificar sesión
        String sessionId = req.getSession(true).getId();

        // PASO 2: Cargar estado L1 → L2
        UserSession sesion = stateManager.obtenerSesion(sessionId);
        if (sesion == null) {
            sesion = new UserSession(sessionId, "anonimo", "GUEST");
        }

        // Inyectar en request
        req.setAttribute("userSession",    sesion);
        req.setAttribute("mdssSessionId",  sessionId);

        // PASO 3: Delegar al Servlet
        // ↓ EL CLIENTE RECIBE SU RESPUESTA AQUÍ DENTRO
        chain.doFilter(request, response);
        // ↑ A partir de aquí el usuario ya tiene sus bytes

        // PASO 4: POST-RESPUESTA → persistir en L1 + Redis
        // Escritura síncrona ~1-2ms que el cliente NO percibe.
        UserSession sesionFinal = (UserSession) req.getAttribute("userSession");
        if (sesionFinal != null) {
            stateManager.persistirSesion(sessionId, sesionFinal);
        }
    }

    @Override
    public void destroy() {
        LOG.info("MdssSessionFilter destruido.");
    }
}
package com.ejemplo.mdss.servlet;

import com.ejemplo.mdss.model.UserSession;
import jakarta.servlet.annotation.WebServlet;
import jakarta.servlet.http.HttpServlet;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;

import java.io.IOException;
import java.io.PrintWriter;

/**
 * SERVLET DE EJEMPLO — Lógica de negocio pura
 * =============================================
 * Este servlet NO sabe nada de Redis, Caffeine ni del patrón MDSS.
 * Solo lee y escribe en request.getAttribute("userSession").
 *
 * CLAVE DE LA ARQUITECTURA: el código legado migra con cambios mínimos.
 *
 *   ANTES:   session.getAttribute("userId")
 *   DESPUÉS: ((UserSession) req.getAttribute("userSession")).getUserId()
 */
@WebServlet("/perfil")
public class PerfilServlet extends HttpServlet {

    @Override
    protected void doGet(HttpServletRequest req, HttpServletResponse resp)
            throws IOException {

        // Leer sesión (cargada por el Filter desde L1 o L2)
        UserSession sesion = (UserSession) req.getAttribute("userSession");

        // ── Lógica de negocio ───────────────────────────────────────────
        sesion.setUltimaPagina("/perfil");
        // ───────────────────────────────────────────────────────────────

        // Devolver sesión modificada → el Filter la persistirá POST-respuesta
        req.setAttribute("userSession", sesion);

        resp.setContentType("application/json;charset=UTF-8");
        PrintWriter out = resp.getWriter();
        out.println("{");
        out.println("  \"userId\":       \"" + sesion.getUserId()       + "\",");
        out.println("  \"username\":     \"" + sesion.getUsername()     + "\",");
        out.println("  \"rol\":          \"" + sesion.getRol()          + "\",");
        out.println("  \"ultimaPagina\": \"" + sesion.getUltimaPagina() + "\",");
        out.println("  \"ultimoAcceso\": \"" + sesion.getUltimoAcceso() + "\"");
        out.println("}");

        // A partir de aquí el Filter escribe en Redis.
        // Este servlet no se entera, ni necesita enterarse.
    }

    @Override
    protected void doPost(HttpServletRequest req, HttpServletResponse resp)
            throws IOException {

        UserSession sesion  = (UserSession) req.getAttribute("userSession");
        String      nuevoRol = req.getParameter("rol");

        if (nuevoRol != null && !nuevoRol.isBlank()) {
            sesion.setRol(nuevoRol);
        }

        req.setAttribute("userSession", sesion);
        resp.setContentType("application/json;charset=UTF-8");
        resp.getWriter().println(
            "{\"resultado\": \"ok\", \"nuevoRol\": \"" + sesion.getRol() + "\"}");
    }
}
package com.ejemplo.mdss.listener;

import com.ejemplo.mdss.cache.SessionStateManager;
import jakarta.servlet.ServletContextEvent;
import jakarta.servlet.ServletContextListener;
import jakarta.servlet.annotation.WebListener;

import java.util.logging.Logger;

/**
 * LIFECYCLE LISTENER
 * ==================
 * Gestiona el ciclo de vida del SessionStateManager:
 *
 *   Arranque → crea el manager → registra en ServletContext
 *   Parada   → cierra el pool de Redis limpiamente
 *
 * Configuración por variables de entorno (sin recompilar el WAR):
 *   REDIS_HOST  → defecto "localhost"
 *   REDIS_PORT  → defecto 6379
 */
@WebListener
public class AppLifecycleListener implements ServletContextListener {

    private static final Logger LOG =
            Logger.getLogger(AppLifecycleListener.class.getName());

    @Override
    public void contextInitialized(ServletContextEvent sce) {
        String redisHost = System.getenv().getOrDefault("REDIS_HOST", "localhost");
        int    redisPort = Integer.parseInt(
                           System.getenv().getOrDefault("REDIS_PORT", "6379"));

        LOG.info("Iniciando SessionStateManager → " + redisHost + ":" + redisPort);

        SessionStateManager manager = new SessionStateManager(redisHost, redisPort);

        // Registrar en contexto para que el Filter lo recupere
        sce.getServletContext().setAttribute("stateManager", manager);

        LOG.info("Aplicación lista. Arquitectura MDSS activa.");
    }

    @Override
    public void contextDestroyed(ServletContextEvent sce) {
        SessionStateManager manager = (SessionStateManager)
                sce.getServletContext().getAttribute("stateManager");

        if (manager != null) {
            manager.cerrar();
            LOG.info("SessionStateManager cerrado limpiamente.");
        }
    }
}
